import {
  FoundationBackend,
  FoundationCatalog,
  FoundationRuntime,
  registerBuiltins,
} from '@source/chat-foundation-v2/server';
import {
  MaterializedResponse,
  type BackendClient,
} from '@source/chat-foundation-v2/contracts';
import { FoundationHttpClient } from '@source/chat-foundation-v2/client';
import {
  ChatShell,
  builtinRegistry,
  type ClientRegistry,
} from '@source/chat-foundation-v2/react';

// Compile and execute against compiled dist/*.js + dist/*.d.ts without tsx.
if (
  ![
    FoundationCatalog,
    FoundationRuntime,
    FoundationBackend,
    registerBuiltins,
    MaterializedResponse,
    FoundationHttpClient,
    ChatShell,
    builtinRegistry,
  ].every(Boolean)
) {
  throw new Error('Missing public export');
}
const catalog = registerBuiltins(
  new FoundationCatalog({
    visible: () => true,
    authorize: async () => ({ scopeKey: 'fixture', context: {} }),
  }),
);
if (!(new FoundationRuntime(catalog) instanceof FoundationRuntime)) {
  throw new Error('Server construction failed');
}
const registry: ClientRegistry = { ...builtinRegistry };
const optionalClient: BackendClient | undefined = undefined;
void registry;
void optionalClient;
console.log('Compiled .d.ts + pure ESM server-check passed');
